cmake_minimum_required(VERSION 3.21)
project(e2e_cpp_sdk VERSION 0.2.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(e2e_cpp_sdk INTERFACE)
add_library(e2e::cpp ALIAS e2e_cpp_sdk)
set_target_properties(e2e_cpp_sdk PROPERTIES EXPORT_NAME cpp)
target_compile_features(e2e_cpp_sdk INTERFACE cxx_std_17)
if(MSVC)
  target_compile_options(e2e_cpp_sdk INTERFACE /utf-8)
elseif(MINGW)
  target_link_options(e2e_cpp_sdk INTERFACE -municode)
endif()
target_include_directories(e2e_cpp_sdk INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
install(TARGETS e2e_cpp_sdk EXPORT e2eCppTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES third_party/nlohmann/json.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/nlohmann)
install(EXPORT e2eCppTargets NAMESPACE e2e:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/e2eCpp)
configure_package_config_file(cmake/e2eCppConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/e2eCppConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/e2eCpp)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/e2eCppConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/e2eCppConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/e2eCppConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/e2eCpp)
install(FILES LICENSE THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/e2eCpp)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/e2eCpp/nlohmann)
include(CTest)
if(BUILD_TESTING)
  add_executable(e2e_cpp_contract_cases tests/contract_cases.cpp)
  target_link_libraries(e2e_cpp_contract_cases PRIVATE e2e::cpp)
  add_executable(e2e_cpp_protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(e2e_cpp_protocol_tests PRIVATE e2e::cpp)
  add_test(NAME cpp-sdk-protocol COMMAND e2e_cpp_protocol_tests)
  set_tests_properties(cpp-sdk-protocol PROPERTIES TIMEOUT 20)
endif()
