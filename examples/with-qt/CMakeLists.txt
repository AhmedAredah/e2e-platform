cmake_minimum_required(VERSION 3.16)
project(e2e_qt_example LANGUAGES CXX)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Test)
option(E2E_QT_WITH_QUICK "Build the optional Qt Quick example" ON)
if(E2E_QT_WITH_QUICK)
  find_package(Qt6 6.2 QUIET COMPONENTS Quick Qml)
endif()
add_subdirectory(../../native/qt native-bridge)
add_executable(e2e_qt_demo main.cpp ui/calculator_window.cpp)
target_compile_features(e2e_qt_demo PRIVATE cxx_std_17)
target_link_libraries(e2e_qt_demo PRIVATE E2E::QtBridge Qt6::Widgets)
if(BUILD_TESTING)
  add_executable(e2e_qt_calculator_tests tests/calculator_tests.cpp ui/calculator_window.cpp)
  set_target_properties(e2e_qt_calculator_tests PROPERTIES AUTOMOC ON)
  target_compile_features(e2e_qt_calculator_tests PRIVATE cxx_std_17)
  target_link_libraries(e2e_qt_calculator_tests PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME qt-calculator-results COMMAND e2e_qt_calculator_tests)
  set_tests_properties(qt-calculator-results PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
if(E2E_QT_WITH_QUICK AND TARGET Qt6::Quick AND TARGET Qt6::Qml)
  add_executable(e2e_qt_quick_demo quick/main.cpp)
  target_compile_features(e2e_qt_quick_demo PRIVATE cxx_std_17)
  target_link_libraries(e2e_qt_quick_demo PRIVATE E2E::QtBridge Qt6::Widgets Qt6::Quick Qt6::Qml)
  add_custom_command(TARGET e2e_qt_quick_demo POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/quick/Main.qml $<TARGET_FILE_DIR:e2e_qt_quick_demo>/Main.qml)
endif()
